\section*{Research Experience}

\blueitem{Predoctoral Researcher \hfill 2019--2025} \\
\sub{University of Granada --- DaSCI Institute}
\begin{itemize}
    \item \textbf{REVEL:} designed a theoretical framework to audit local linear explanations for robustness, conciseness, and prescriptivity; evaluated on deep learning image classification benchmarks.
    \item \textbf{X-SHIELD:} proposed an explanation-driven regularization family that selects input features to hide during training, improving model performance and explainability simultaneously.
    \item \textbf{STOOD-X:} developed a two-stage explainable OOD detection method combining Wilcoxon-Mann-Whitney nonparametric testing with concept-based visual explanations.
    \item \textbf{COVIDGR:} co-designed a balanced COVID-19 chest X-ray dataset across all severity levels and the COVID-SDNet methodology.
\end{itemize}

\section*{Open-Source Software \& Datasets}

\blueitem{Software}
\begin{itemize}
    \item \textbf{S-ReVel} --- audits local linear explanations (e.g.\ LIME) for robustness, conciseness, and prescriptivity. \href{https://github.com/ari-dasci/S-ReVel}{github.com/ari-dasci/S-ReVel}. AGPL-3.0.
    \item \textbf{S-STOOD-X} --- explainable OOD detection via Wilcoxon-Mann-Whitney testing and concept-based visual explanations. \href{https://github.com/ari-dasci/S-STOOD-X}{github.com/ari-dasci/S-STOOD-X}. GPL-3.0.
    \item \textbf{S-T-SHIELD} --- explanation-driven regularization that hides input features during training to improve performance and explainability. \href{https://github.com/ari-dasci/S-T-SHIELD}{github.com/ari-dasci/S-T-SHIELD}. GPL-3.0.
\end{itemize}

\blueitem{Datasets}
\begin{itemize}
    \item \textbf{COVIDGR} --- COVID-19 chest X-ray dataset with fine-grained classification across severity levels, from positive RT-PCR with no symptoms to severe cases. \href{https://github.com/ari-dasci/OD-covidgr}{github.com/ari-dasci/OD-covidgr}.
    \item \textbf{MetalDAM} --- metallographic image dataset for microstructural segmentation in steel industry quality control. \href{https://github.com/ari-dasci/OD-MetalDAM}{github.com/ari-dasci/OD-MetalDAM}.
\end{itemize}
